## 一、痛点深度研究报告

### 1.1 核心痛点

**iOS系统闹钟音量与铃声音量绑定，无法为每个闹钟设置独立音量。**

这是自iOS诞生以来持续存在的用户痛点，用户抱怨跨度超过6年（2019-2026），Apple Community相关帖子累计超过1000条，Reddit相关讨论帖数百条。

### 1.5 技术可行性突破点: iOS 26 AlarmKit

2025年WWDC，Apple发布了**AlarmKit框架**（iOS 26+），这是革命性的变化：

- ✅ 第三方应用可以创建**系统级闹钟**
- ✅ 闹钟**突破静音模式和专注模式**
- ✅ 支持**锁屏和灵动岛**显示
- ✅ 支持**无限数量闹钟**和重复选项
- ✅ 闹钟不会因应用更新或手机重启而失效
- ✅ 支持**自定义声音**

结合AVAudioPlayer的独立音量控制能力，**per-alarm独立音量控制首次成为可能**。

#### 2.3.2 闹钟调度器 — AlarmScheduler.swift

```swift
import Foundation
import AlarmKit

@MainActor
final class AlarmScheduler: ObservableObject {
    static let shared = AlarmScheduler()

    private let alarmManager = AlarmManager.shared

    private init() {}

    func requestAuthorization() async throws -> AlarmManager.AuthorizationStatus {
        try await alarmManager.requestAuthorization()
    }

    func scheduleAlarm(_ model: AlarmModel) async throws -> UUID {
        let weekdays = convertToAlarmKitWeekdays(model.selectedDays)

        let schedule = Alarm.Schedule.relative(
            Alarm.Schedule.Relative(
                time: Alarm.Schedule.Relative.Time(hour: model.hour, minute: model.minute),
                repeats: .weekly(weekdays)
            )
        )

        let countdownDuration = Alarm.CountdownDuration(
            preAlert: nil,
            postAlert: TimeInterval(model.snoozeDuration)
        )

        let alertPresentation = AlarmPresentation.Alert(
            title: LocalizedStringResource(stringLiteral: model.name),
            stopButton: AlarmButton(
                text: "Stop",
                textColor: .white,
                systemImageName: "stop.fill"
            ),
            secondaryButton: AlarmButton(
                text: "Snooze",
                textColor: .white,
                systemImageName: "repeat.circle.fill"
            ),
            secondaryButtonBehavior: .countdown
        )

        let countdownPresentation = AlarmPresentation.Countdown(
            title: LocalizedStringResource(stringLiteral: "Snoozing..."),
            pauseButton: AlarmButton(
                text: "Snooze",
                textColor: .white,
                systemImageName: "repeat.circle.fill"
            )
        )

        let presentation = AlarmPresentation(
            alert: alertPresentation,
            countdown: countdownPresentation
        )

        let metadata = VolarmMetadata(
            alarmName: model.name,
            volume: model.volume,
            soundIdentifier: model.soundIdentifier,
            isGradualVolume: model.isGradualVolume,
            gradualDuration: model.gradualDuration,
            label: model.label
        )

        let attributes = AlarmAttributes(
            presentation: presentation,
            metadata: metadata,
            tintColor: volumeToColor(model.volume)
        )

        let sound = resolveSound(model.soundIdentifier)

        let configuration = AlarmManager.AlarmConfiguration(
            countdownDuration: countdownDuration,
            schedule: schedule,
            attributes: attributes,
            secondaryIntent: nil,
            sound: sound
        )

        let alarmID = try await alarmManager.schedule(
            id: model.id,
            configuration: configuration
        )

        return alarmID
    }

    func stopAlarm(id: UUID) async throws {
        try await alarmManager.stop(id: id)
    }

    func snoozeAlarm(id: UUID) async throws {
        try await alarmManager.snooze(id: id)
    }

    private func convertToAlarmKitWeekdays(_ days: Set<Int>) -> [Alarm.Schedule.Relative.Recurrence.Weekday] {
        var weekdays: [Alarm.Schedule.Relative.Recurrence.Weekday] = []
        let mapping: [Int: Alarm.Schedule.Relative.Recurrence.Weekday] = [
            0: .sunday, 1: .monday, 2: .tuesday, 3: .wednesday,
            4: .thursday, 5: .friday, 6: .saturday
        ]
        for day in days {
            if let weekday = mapping[day] {
                weekdays.append(weekday)
            }
        }
        return weekdays
    }

    private func volumeToColor(_ volume: Float) -> Color {
        switch volume {
        case 0..<0.3: return .blue
        case 0.3..<0.6: return .green
        case 0.6..<0.8: return .orange
        default: return .red
        }
    }

    private func resolveSound(_ identifier: String) -> AlarmSound {
        if identifier == "default" {
            return .default
        }
        return .named(identifier)
    }
}
```

#### 2.3.3 音量管理器 — VolumeManager.swift

```swift
import Foundation
import AVFAudio

@MainActor
final class VolumeManager {
    static let shared = VolumeManager()

    private var audioPlayer: AVAudioPlayer?
    private var currentVolume: Float = 1.0
    private var gradualTimer: Timer?

    private init() {}

    func configureAudioSession() {
        let session = AVAudioSession.sharedInstance()
        do {
            try session.setCategory(.playback, mode: .default, options: [.mixWithOthers])
            try session.setActive(true)
        } catch {
            print("Failed to configure audio session: \(error)")
        }
    }

    func playAlarmSound(soundIdentifier: String, volume: Float, isGradual: Bool = false, gradualDuration: Int = 30) {
        configureAudioSession()

        guard let url = resolveSoundURL(soundIdentifier) else { return }

        do {
            audioPlayer = try AVAudioPlayer(contentsOf: url)
            audioPlayer?.numberOfLoops = -1
            audioPlayer?.prepareToPlay()

            if isGradual {
                audioPlayer?.volume = 0.0
                audioPlayer?.play()
                startGradualVolumeIncrease(targetVolume: volume, duration: TimeInterval(gradualDuration))
            } else {
                audioPlayer?.volume = volume
                audioPlayer?.play()
            }

            currentVolume = volume
        } catch {
            print("Failed to play alarm sound: \(error)")
        }
    }

    func stopAlarmSound() {
        gradualTimer?.invalidate()
        gradualTimer = nil
        audioPlayer?.stop()
        audioPlayer = nil
    }

    func setVolume(_ volume: Float) {
        currentVolume = volume
        audioPlayer?.setVolume(volume, fadeDuration: 0.3)
    }

    private func startGradualVolumeIncrease(targetVolume: Float, duration: TimeInterval) {
        let steps = Int(duration)
        let volumeStep = targetVolume / Float(steps)
        var currentStep = 0

        gradualTimer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] timer in
            guard let self = self else {
                timer.invalidate()
                return
            }
            currentStep += 1
            let newVolume = min(volumeStep * Float(currentStep), targetVolume)
            self.audioPlayer?.setVolume(newVolume, fadeDuration: 0.8)

            if currentStep >= steps {
                timer.invalidate()
                self.gradualTimer = nil
            }
        }
    }

    private func resolveSoundURL(_ identifier: String) -> URL? {
        if identifier == "default" {
            return Bundle.main.url(forResource: "default_alarm", withExtension: "mp3")
        }
        if identifier.hasPrefix("custom_") {
            let fileName = String(identifier.dropFirst(7))
            return getCustomSoundURL(fileName: fileName)
        }
        return Bundle.main.url(forResource: identifier, withExtension: "mp3")
    }

    private func getCustomSoundURL(fileName: String) -> URL? {
        let documentsDir = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
        return documentsDir.appendingPathComponent("sounds/\(fileName).mp3")
    }
}
```

```swift
struct VolumePreviewBar: View {
    let volume: Float

    var body: some View {
        HStack(spacing: 2) {
            ForEach(0..<20, id: \.self) { index in

                    .fill(barColor(for: index))
                    .frame(height: barHeight(for: index))

        case 0..<0.3: return .blue

        case 0.6..<0.8: return .orange

        }
    }
```

```swift
import SwiftUI
import SwiftData

struct AlarmListView: View {
    @Environment(\.modelContext) private var modelContext
    @Query(sort: \AlarmModel.hour) private var alarms: [AlarmModel]
    @State private var showingAddAlarm = false
    @State private var authorizationStatus: AlarmManager.AuthorizationStatus = .notDetermined

    var body: some View {
        NavigationStack {
            Group {
                if alarms.isEmpty {
                    emptyStateView
                } else {
                    alarmList
                }
            }
            .navigationTitle("Volarm")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button {
                        showingAddAlarm = true
                    } label: {
                        Image(systemName: "plus")
                    }
                }
            }
            .sheet(isPresented: $showingAddAlarm) {
                AlarmEditView(mode: .create)
            }
            .task {
                await checkAuthorization()
            }
        }
    }

    private var emptyStateView: some View {
        ContentUnavailableView {
            Label("No Alarms", systemImage: "alarm")
        } description: {
            Text("Tap + to create your first alarm with custom volume")
        } actions: {
            Button("Create Alarm") {
                showingAddAlarm = true
            }
            .buttonStyle(.borderedProminent)
        }
    }

    private var alarmList: some View {
        List {
            ForEach(alarms) { alarm in
                NavigationLink(destination: AlarmEditView(mode: .edit(alarm))) {
                    AlarmRowView(alarm: alarm)
                }
            }
            .onDelete(perform: deleteAlarms)
        }
        .listStyle(.insetGrouped)
    }

    private func deleteAlarms(offsets: IndexSet) {
        withAnimation {
            for index in offsets {
                let alarm = alarms[index]
                Task {
                    try? await AlarmScheduler.shared.stopAlarm(id: alarm.id)
                }
                modelContext.delete(alarm)
            }
        }
    }

    private func checkAuthorization() async {
        do {
            authorizationStatus = try await AlarmScheduler.shared.requestAuthorization()
        } catch {
            print("Authorization error: \(error)")
        }
    }
}
```

```swift
                try await AlarmScheduler.shared.scheduleAlarm(alarm)
                dismiss()
            } catch {
                print("Save error: \(error)")
            }
```

### 2.4 编写规则

| 规则 | 说明 |
|------|------|
| 架构模式 | MVVM + Repository Pattern，View不直接访问数据层 |
| 命名规范 | Model后缀Model，View后缀View，Manager后缀Manager |
| 并发安全 | 所有AlarmKit调用必须在@MainActor上下文中 |
| 错误处理 | 使用async/await + try/catch，不使用optional try |
| 音量范围 | Float 0.0-1.0，UI显示为0%-100% |
| 音量颜色 | 0-30%蓝色(轻柔)、30-60%绿色(适中)、60-80%橙色(较响)、80-100%红色(最响) |
| 数据验证 | 闹钟名称不能为空、至少选择一天、音量范围0-1 |
| 闹钟生命周期 | 创建→调度→触发→停止/贪睡→重新调度 |
| 渐进音量 | 从0开始，按设定时长线性增长到目标音量 |
| 音频会话 | 使用.playback类别，确保闹钟声音不受系统音量影响 |
